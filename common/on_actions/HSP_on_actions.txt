on_actions = {
	on_startup = {
		effect = {
			# 1936 start on the HSP tree: only 전국 파업 (SPR_national_strikes_3) and 구식 군대 remain,
			# and division templates are unlocked (vanilla 05_lar_on_actions.txt locks them; this file runs after it).
			# HSP_return_to_historical_spain puts the vanilla spirits back if the player leaves the HSP tree.
			# Same conditions as HSP_spanish_focus's country weight; the tree may not be assigned yet here.
			SPR = {
				if = {
					limit = {
						date < 1936.1.2
						has_dlc = "La Resistance"
					}
					remove_ideas = {
						SPA_carlism_1
						SPR_military_disloyalty
						SPR_political_violence
					}
					add_ideas = HSP_old_fashioned_army_01
					country_lock_all_division_template = no
				}
			}
		}
	}

	# ROOT = winner, FROM = loser
	on_civil_war_end = {
		effect = {
			# The Republic wins against the fascist uprising (HSP_start_fascist_uprising)
			if = {
				limit = {
					original_tag = SPR
					has_country_flag = HSP_fascist_uprising_flag
					NOT = { has_country_flag = HSP_fascists_defeated_flag }
					NOT = { has_government = fascism }
					FROM = {
						original_tag = SPR
						has_government = fascism
					}
				}
				HSP_fascists_defeated = yes
				news_event = { id = HSP_spain_news.4 hours = 6 }
			}
		}
	}
}
