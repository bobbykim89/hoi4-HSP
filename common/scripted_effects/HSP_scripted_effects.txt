# Scope: the country that becomes the viceroyalty. ROOT = Spain.
# Becomes Spain's subject; call HSP_join_viceroyalty on each country it absorbs afterwards.
HSP_form_viceroyalty = {
	save_event_target_as = HSP_viceroy
	set_country_flag = HSP_viceroyalty_flag
	if = {
		limit = { NOT = { is_subject_of = ROOT } }
		ROOT = { puppet = PREV }
	}
}

# Scope: a country joining the viceroyalty saved as event_target:HSP_viceroy.
# The viceroyalty gets cores on all its core states and annexes it if it still exists.
HSP_join_viceroyalty = {
	every_state = {
		limit = { is_core_of = PREV }
		add_core_of = event_target:HSP_viceroy
	}
	if = {
		limit = { exists = yes }
		event_target:HSP_viceroy = {
			annex_country = {
				target = PREV
				transfer_troops = yes
			}
		}
	}
}

# Scope: a viceroyalty or a Latin Sphere member. ROOT = Spain. Its generals change nationality, then Spain annexes it and takes its cores.
HSP_annex_viceroyalty = {
	every_state = {
		limit = { is_core_of = PREV }
		add_core_of = ROOT
	}
	every_unit_leader = {
		set_nationality = ROOT
	}
	ROOT = {
		annex_country = {
			target = PREV
			transfer_troops = yes
		}
	}
}

# Scope: the overlord. Makes event_target:HSP_union_target its personal union (No Step Back),
# or a regular puppet without that DLC.
HSP_make_personal_union = {
	if = {
		limit = { has_dlc = "No Step Back" }
		set_autonomy = {
			target = event_target:HSP_union_target
			autonomy_state = autonomy_personal_union
		}
	}
	else = {
		puppet = event_target:HSP_union_target
	}
}

# Timeout of HSP_fascist_plot_mission. The historical Nationalist states rise, except those where
# HSP_maintain_order was taken. Template: vanilla events/Spain.txt spain.1 option b.
HSP_start_fascist_uprising = {
	every_owned_state = {
		limit = {
			OR = {
				state = 171
				state = 172
				state = 174
				state = 176
				state = 791
				state = 788
				state = 166
				state = 169
				state = 170
				state = 789
				state = 177
				state = 178
				state = 290
				state = 297
				state = 699
			}
			is_controlled_by = ROOT
			NOT = { has_state_flag = HSP_order_maintained_flag }
		}
		set_state_flag = HSP_fascist_state_flag
	}
	if = {
		limit = {
			NOT = {
				any_owned_state = { has_state_flag = HSP_fascist_state_flag }
			}
		}
		# every state was pacified: the plot collapses without a war
		HSP_fascists_defeated = yes
		news_event = { id = HSP_spain_news.3 hours = 6 }
	}
	else = {
		set_country_flag = HSP_fascist_uprising_flag
		set_variable = { HSP_uprising_size = 0.1 }
		every_owned_state = {
			limit = { has_state_flag = HSP_fascist_state_flag }
			ROOT = { add_to_variable = { HSP_uprising_size = 0.03 } }
		}
		start_civil_war = {
			ideology = fascism
			size = var:HSP_uprising_size
			states = all
			states_filter = { has_state_flag = HSP_fascist_state_flag }
			# Spain keeps every character; the Nationalists are sent over by name below
			# (by default the game hands the rebels a share of everyone, left-wing advisors included)
			keep_all_characters = yes
			save_event_target_as = HSP_fascist_rebels
			set_cosmetic_tag = SPR_civil_war
			set_country_flag = HSP_fascist_rebels_flag
			set_stability = 0.7
			set_war_support = 0.8
		}
		HSP_send_fascists_to_rebels = yes
		if = {
			limit = { has_idea = HSP_fascist_conspiracy }
			remove_ideas = HSP_fascist_conspiracy
		}
		news_event = { id = HSP_spain_news.2 hours = 6 }
	}
}

		type = building_target
		id = infrastructure
		value = 25
	}
	add_ai_strategy = {
		type = alliance
		id = ROOT
		value = 200
	}
}

		}

# Scope: the country that rises. ROOT = Spain. A communist uprising sized by communist support (25-60%);
# Spain joins the rebels and makes them its puppet (vanilla lar_spain.49 and HSP_spain_events.36 pattern).

	set_temp_variable = { HSP_revolt_size = party_popularity@communism }
	clamp_temp_variable = {
		var = HSP_revolt_size
		min = 0.25
		max = 0.6
	}
	start_civil_war = {
		ideology = communism
		size = var:HSP_revolt_size
		ROOT = {
			add_to_war = {
				targeted_alliance = PREV
				enemy = event_target:HSP_revolution_target
				hostility_reason = asked_to_join
			}
			puppet = {
				target = PREV
				end_wars = no
				end_civil_wars = no
			}
		}
	}
}

# Scope: the country that rises. ROOT = Spain. An anarchist uprising in the states nearest Spain; Spain joins
# the rebels' war. Anarchists take no puppets: the commune stays free (Portuguese anarchism, HSP_ignite_anarchist_uprising).
HSP_start_anarchist_uprising = {
	save_event_target_as = HSP_revolution_target
	if = {
		limit = {
			any_controlled_state = {
				is_capital = no
				any_neighbor_state = { is_controlled_by = ROOT }
			}
		}
		every_controlled_state = {
			limit = {
				is_capital = no
				any_neighbor_state = { is_controlled_by = ROOT }
			}
			set_state_flag = HSP_anarchist_uprising_state
		}
	}

# Fascist path: timeout of HSP_nationalist_plot_mission, or stability 0% with fascism above 60%.
# Spain becomes Nationalist under Franco and keeps the historical Nationalist zone plus the states won over
# (HSP_win_over_state); the Republic revolts with the rest. The player keeps the HSP tree.
HSP_start_nationalist_uprising = {
	set_country_flag = HSP_nationalist_uprising_flag
	if = {
		limit = { has_active_mission = HSP_nationalist_plot_mission }
		remove_mission = HSP_nationalist_plot_mission
	}
	if = {
		limit = { has_idea = HSP_fascist_conspiracy }
		remove_ideas = HSP_fascist_conspiracy
	}
	HSP_retire_republican_characters = yes
	HSP_retire_communist_characters = yes
	set_politics = {
		ruling_party = fascism
		elections_allowed = no
	}
	if = {
		limit = { has_character = SPA_francisco_franco }
		add_country_leader_role = {
			character = SPA_francisco_franco
			country_leader = {
				ideology = fascism_ideology
				traits = { caudillo_1 }
				expire = "1965.1.1"
			}
			promote_leader = yes
		}
	}
	set_cosmetic_tag = SPR_civil_war
	# the historical Nationalist zone (vanilla spain.1's 1936 split, plus Ifni) always rises with us
	every_owned_state = {
		limit = {
			OR = {
				state = 166
				state = 169
				state = 170
				state = 171
				state = 172
				state = 174
				state = 176
				state = 177
				state = 178
				state = 290
				state = 297
				state = 699
				state = 783
				state = 788
				state = 789
				state = 791
			}
		}
		set_state_flag = HSP_won_over_flag
	}
	# the capital must stay with us: Burgos (the Nationalist capital) or another state we hold
	if = {
		limit = { NOT = { capital_scope = { has_state_flag = HSP_won_over_flag } } }
		if = {
			limit = { 176 = { has_state_flag = HSP_won_over_flag is_owned_by = ROOT } }
			set_capital = { state = 176 }
		}
		else = {
			random_owned_state = {
				limit = { has_state_flag = HSP_won_over_flag }
				ROOT = { set_capital = { state = PREV } }
			}
		}
	}
	if = {
		limit = { democratic < 0.05 }
		add_popularity = {
			ideology = democratic
			popularity = 0.1
		}
	}
	start_civil_war = {
		ideology = democratic
		size = 0.5
		states = all
		states_filter = {
			NOT = { has_state_flag = HSP_won_over_flag }
		}
		keep_all_characters = yes # the Republican and communist figures are retired just above
		set_cosmetic_tag = SPR_republican_spain
		set_country_flag = HSP_republican_rebels_flag
	}
	news_event = { id = HSP_spain_news.10 hours = 6 }
}

# HSP_nat_restore_the_monarchy: a king reigns over a neutrality government while Franco keeps the army.
