# Scope: the country that becomes the viceroyalty. ROOT = Spain.
# Becomes Spain's subject; call HSP_join_viceroyalty on each country it absorbs afterwards.
HSP_form_viceroyalty = {
	save_event_target_as = HSP_viceroy
	set_country_flag = HSP_viceroyalty_flag
	if = {
		limit = { NOT = { is_subject_of = ROOT } }
		ROOT = { puppet = PREV }
	}
}

# Scope: a country joining the viceroyalty saved as event_target:HSP_viceroy.
# The viceroyalty gets cores on all its core states and annexes it if it still exists.
HSP_join_viceroyalty = {
	every_state = {
		limit = { is_core_of = PREV }
		add_core_of = event_target:HSP_viceroy
	}
	if = {
		limit = { exists = yes }
		event_target:HSP_viceroy = {
			annex_country = {
				target = PREV
				transfer_troops = yes
			}
		}
	}
}

# Scope: a viceroyalty or a Latin Sphere member. ROOT = Spain. Its generals change nationality, then Spain annexes it and takes its cores.
HSP_annex_viceroyalty = {
	every_state = {
		limit = { is_core_of = PREV }
		add_core_of = ROOT
	}
	every_unit_leader = {
		set_nationality = ROOT
	}
	ROOT = {
		annex_country = {
			target = PREV
			transfer_troops = yes
		}
	}
}

# Scope: the overlord. Makes event_target:HSP_union_target its personal union (No Step Back),
# or a regular puppet without that DLC.
HSP_make_personal_union = {
	if = {
		limit = { has_dlc = "No Step Back" }
		set_autonomy = {
			target = event_target:HSP_union_target
			autonomy_state = autonomy_personal_union
		}
	}
	else = {
		puppet = event_target:HSP_union_target
	}
}

# The fascist plot is over, by HSP_eliminate_the_fascists or by winning the uprising (HSP_on_actions).
# Unlocks the choice between democracy, communism and anarchism.
HSP_fascists_defeated = {
	custom_effect_tooltip = HSP_fascists_defeated_tt
	hidden_effect = {
		set_country_flag = HSP_fascists_defeated_flag
		set_global_flag = scw_over
		if = {
			limit = { has_idea = HSP_fascist_conspiracy }
			remove_ideas = HSP_fascist_conspiracy
		}
		if = {
			limit = { has_active_mission = HSP_fascist_plot_mission }
			remove_mission = HSP_fascist_plot_mission
		}
	}
}

# Timeout of HSP_fascist_plot_mission. The historical Nationalist states rise, except those where
# HSP_maintain_order was taken. Template: vanilla events/Spain.txt spain.1 option b.
HSP_start_fascist_uprising = {
	every_owned_state = {
		limit = {
			OR = {
				state = 171
				state = 172
				state = 174
				state = 176
				state = 791
				state = 788
				state = 166
				state = 169
				state = 170
				state = 789
				state = 177
				state = 178
				state = 290
				state = 297
				state = 699
			}
			is_controlled_by = ROOT
			NOT = { has_state_flag = HSP_order_maintained_flag }
		}
		set_state_flag = HSP_fascist_state_flag
	}
	if = {
		limit = {
			NOT = {
				any_owned_state = { has_state_flag = HSP_fascist_state_flag }
			}
		}
		# every state was pacified: the plot collapses without a war
		HSP_fascists_defeated = yes
		news_event = { id = HSP_spain_news.3 hours = 6 }
	}
	else = {
		set_country_flag = HSP_fascist_uprising_flag
		set_variable = { HSP_uprising_size = 0.1 }
		every_owned_state = {
			limit = { has_state_flag = HSP_fascist_state_flag }
			ROOT = { add_to_variable = { HSP_uprising_size = 0.03 } }
		}
		start_civil_war = {
			ideology = fascism
			size = var:HSP_uprising_size
			states = all
			states_filter = { has_state_flag = HSP_fascist_state_flag }
			# Franco (legacy id 1108) has no sympathies trait
			keep_unit_leaders_trigger = {
				NOT = { has_trait = trait_SPA_nationalist_sympathies }
				NOT = { has_id = 1108 }
			}
			set_cosmetic_tag = SPR_civil_war
			set_country_flag = HSP_fascist_rebels_flag
			set_stability = 0.7
			set_war_support = 0.8
		}
		if = {
			limit = { has_idea = HSP_fascist_conspiracy }
			remove_ideas = HSP_fascist_conspiracy
		}
		news_event = { id = HSP_spain_news.2 hours = 6 }
	}
}

# Scope: a region released as Spain's puppet (democracy split: GLC, NAV, CAT, MOR).
# Development spirit, a civilian factory and infrastructure, the HSP_regional_focus development tree
# (common/national_focus/HSP_regional.txt), and an AI that builds industry.
HSP_setup_regional_puppet = {
	add_ideas = HSP_regional_development
	load_focus_tree = {
		tree = HSP_regional_focus
		keep_completed = no
	}
	capital_scope = {
		add_extra_state_shared_building_slots = 1
		add_building_construction = {
			type = industrial_complex
			level = 1
			instant_build = yes
		}
	}
	every_owned_state = {
		limit = { infrastructure < 5 }
		add_building_construction = {
			type = infrastructure
			level = 1
			instant_build = yes
		}
	}
	add_ai_strategy = {
		type = building_target
		id = industrial_complex
		value = 50
	}
	add_ai_strategy = {
		type = building_target
		id = infrastructure
		value = 25
	}
	add_ai_strategy = {
		type = alliance
		id = ROOT
		value = 200
	}
}

}

# Scope: a country joining Spain's faction. ROOT = Spain. If Spain is not the leader (the Allies),
# its faction leader adds the country.
HSP_join_spanish_faction = {
	if = {
		limit = { ROOT = { is_faction_leader = yes } }
		ROOT = { add_to_faction = PREV }
	}
	else = {
		random_country = {
			limit = {
				is_faction_leader = yes
				is_in_faction_with = ROOT
			}
			add_to_faction = PREV
		}
	}
	add_ai_strategy = {
		type = alliance
		id = ROOT
		value = 200
	}
}

# Scope: the country that rises. ROOT = Spain. A communist uprising sized by communist support (25-60%);
# Spain joins the rebels and makes them its puppet (vanilla lar_spain.49 and HSP_spain_events.36 pattern).
HSP_start_communist_uprising = {
	save_event_target_as = HSP_revolution_target
	set_temp_variable = { HSP_revolt_size = party_popularity@communism }
	clamp_temp_variable = {
		var = HSP_revolt_size
		min = 0.25
		max = 0.6
	}
	start_civil_war = {
		ideology = communism
		size = var:HSP_revolt_size
		ROOT = {
			add_to_war = {
				targeted_alliance = PREV
				enemy = event_target:HSP_revolution_target
				hostility_reason = asked_to_join
			}
			puppet = {
				target = PREV
				end_wars = no
				end_civil_wars = no
			}
		}
	}
}

# HSP_com_portuguese_revolution. A communist Portugal is offered a union instead (HSP_spain_events.46).
HSP_start_portuguese_revolution = {
	random_country = {
		limit = {
			original_tag = POR
			NOT = { is_subject_of = ROOT }
		}
		if = {
			limit = { has_government = communism }
			country_event = { id = HSP_spain_events.46 hours = 6 }
		}
		else = {
			HSP_start_communist_uprising = yes
			news_event = { id = HSP_spain_news.5 hours = 6 }
		}
	}
}
