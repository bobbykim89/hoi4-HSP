# Otto von Habsburg (vanilla HUN_otto_von_habsburg) can be crowned King of Spain:
# Hungary or Austria hasn't taken him, and he leads no other country.
HSP_otto_available = {
	custom_trigger_tooltip = {
		tooltip = HSP_otto_available_tt
		country_exists = HUN
		NOT = {
			has_global_flag = HUN_otto_von_habsburg_has_joined_hungary_flag
			has_global_flag = AUS_otto_von_habsburg_has_joined_austria_flag
		}
		NOT = {
			any_other_country = {
				has_country_leader = {
					character = HUN_otto_von_habsburg
					ruling_only = yes
				}
			}
		}
	}
}

# Alfonso XIII is alive and hasn't been crowned in France (vanilla FRA_the_legitimate_heir makes him Alphonse I).

		}
	}
}

# Scope: a Latin American country. AI accepts Spanish leadership (faction invite) when Spain's army is
# stronger and it likes Spain.
HSP_accepts_spanish_leadership = {
	custom_trigger_tooltip = {
		tooltip = HSP_accepts_spanish_leadership_tt
		strength_ratio = {
			tag = SPR
			ratio < 1
		}
		has_opinion = {
			target = SPR
			value > 50
		}
	}
}

# Scope: a released region (GLC, CAT). ROOT = Spain. Our puppet, or our ally once the Iberian Defense Council
# has set it free, and not at war with us (HSP_federal_claims_cat).
HSP_is_federal_partner = {
	custom_trigger_tooltip = {
		tooltip = HSP_is_federal_partner_tt
		exists = yes
		OR = {
			is_subject_of = ROOT
			is_in_faction_with = ROOT
		}
		NOT = { has_war_with = ROOT }
	}
}

# Limited wars (HSP_federal_claims_cat). Scope: the target. ROOT = Spain.
HSP_can_start_limited_war = {
	NOT = { is_ally_with = ROOT }
	NOT = { has_war_with = ROOT }
	NOT = { has_non_aggression_pact_with = ROOT }
	NOT = { is_guaranteed_by = ROOT }
	# The peace covers the target's whole side, so it must not end a war we were already fighting
	custom_trigger_tooltip = {
		tooltip = HSP_target_side_not_at_war_tt
		NOT = { any_allied_country = { has_war_with = ROOT } }
		NOT = {
			any_country = {
				has_war_with = ROOT
				OR = {
					has_guaranteed = PREV
					has_war_together_with = PREV
					is_in_faction_with = PREV
				}
			}
		}
		NOT = {
			AND = {
				is_subject = yes
				overlord = { has_war_with = ROOT }
			}
		}
	}
	custom_trigger_tooltip = {
		tooltip = HSP_no_limited_war_running_tt
		ROOT = { NOT = { has_variable = HSP_limited_war_target } }
	}
}

# Limited wars. Scope: a country at war with the target's side. ROOT = Spain.
HSP_is_on_our_limited_war_side = {
	OR = {
		tag = ROOT
		is_subject_of = ROOT
		AND = {
			is_in_faction_with = ROOT
			ROOT = { is_faction_leader = yes }
		}
	}
}

# Limited wars. Scope: Spain. Every recorded state is held by Spain or one of its subjects.
HSP_limited_war_won = {

		array = HSP_limited_war_states
		OR = {
			is_controlled_by = ROOT
			controller = { is_subject_of = ROOT }
		}
	}
}
