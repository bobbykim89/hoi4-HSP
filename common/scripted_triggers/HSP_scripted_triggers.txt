# Otto von Habsburg (vanilla HUN_otto_von_habsburg) can be crowned King of Spain:
# Hungary or Austria hasn't taken him, and he leads no other country.
HSP_otto_available = {
	custom_trigger_tooltip = {
		tooltip = HSP_otto_available_tt
		country_exists = HUN
		NOT = {
			has_global_flag = HUN_otto_von_habsburg_has_joined_hungary_flag
			has_global_flag = AUS_otto_von_habsburg_has_joined_austria_flag
		}
		NOT = {
			any_other_country = {
				has_country_leader = {
					character = HUN_otto_von_habsburg
					ruling_only = yes
				}
			}
		}
	}
}

# Alfonso XIII is alive and hasn't been crowned in France (vanilla FRA_the_legitimate_heir makes him Alphonse I).
HSP_alfonso_available = {
	custom_trigger_tooltip = {
		tooltip = HSP_alfonso_available_tt
		NOT = { has_global_flag = SPB_alfonso_xiii_dead }
		NOT = {
			FRA = { has_completed_focus = FRA_the_legitimate_heir }
		}
	}
}

# Scope: a Latin American country. AI accepts Spanish leadership (faction invite) when Spain's army is
# stronger and it likes Spain.
HSP_accepts_spanish_leadership = {
	custom_trigger_tooltip = {
		tooltip = HSP_accepts_spanish_leadership_tt
		strength_ratio = {
			tag = SPR
			ratio < 1
		}
		has_opinion = {
			target = SPR
			value > 50
		}
	}
}

# Scope: a country named in a viceroyalty decision. ROOT = Spain.
# It must be in Spain's faction or Spain's subject; a country that no longer exists is skipped.
HSP_viceroyalty_candidate = {
	OR = {
		exists = no
		is_subject_of = ROOT
		is_in_faction_with = ROOT
	}
}

# Scope: Hungary or Austria. AI accepts the Spanish Habsburg crown when Spain's army is stronger and
# Spain fields more than 500k men.
HSP_accepts_habsburg_crown = {
	custom_trigger_tooltip = {
		tooltip = HSP_accepts_habsburg_crown_tt
		strength_ratio = {
			tag = SPR
			ratio < 1
		}
		SPR = {
			has_army_manpower = { size > 500000 }
		}
	}
}
